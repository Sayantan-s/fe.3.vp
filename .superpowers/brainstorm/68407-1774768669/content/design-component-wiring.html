RightPanel — Player Integration

@rightPanel/page.tsx
// "use client"
// Player is lazy loaded — Three.js bundle stays out of initial chunk

const LazyPlayer = lazy(() =>
import("@/components/stories/organisms/player")
);

export default function RightPanelPage() {
const { videoUrl, bg, padding, rounding, isLoading, error } = useVideoDataState();

if (isLoading) return <RightPanelSkeleton />;
if (error) return <RightPanelError error={error} />;

return (
<Suspense fallback={<PlayerSkeleton />}>
<LazyPlayer.Canvas
padding={padding}
rounding={rounding}
>
<LazyPlayer.Canvas.Background backgroundSrc={bg} />
<LazyPlayer.Canvas.Video videoSrc={videoUrl} />
<PlaybackBar />
</LazyPlayer.Canvas>
</Suspense>
);
}

LeftPanel — Controls Wiring

ControlsPanel (no props — all from context)
// No props. No prop drilling. Everything from context.

function ControlsPanel() {
const { padding, rounding } = useVideoDataState();
const { setPadding, setRounding } = useVideoDataActions();

return (
<RangeSlider label="Padding" value={padding} onValueChange={setPadding} />
<RangeSlider label="Rounding" value={rounding} onValueChange={setRounding} />
);
}

// setPadding/setRounding are stable refs (useCallback)
// ControlsPanel re-renders only when padding/rounding VALUES change
// Other components reading only actions never re-render from state changes

Data Flow Diagram

Slider change → Player update (no prop drilling)
1 User drags padding slider in ControlsPanel
↓
2 Calls setPadding(20) from useVideoDataActions
↓
3 VideoDataProvider updates state immediately (optimistic)
↓
4 VideoDataStateContext value changes → { padding: 20 }
↓ (context subscription)
5 RightPanel re-reads padding → passes as controlled prop to Player.Canvas
↓
6 Player.Canvas receives new padding → useControlledSync → renders
